# RuleRequest

## Example Usage

```typescript
import { RuleRequest } from "@formance/formance-sdk/sdk/models/reconciliation";

let value: RuleRequest = {
  name: "<value>",
  schedule: {
    expr: "*/15 * * * *",
    kind: "cron",
    safetyMargin: "30s",
    tz: "UTC",
  },
  templateKind: "ledger_invariant",
  templateSpec: {
    "key": "<value>",
  },
};
```
