# Account

An account held at a payment provider, surfaced through a connector

## Example Usage

```typescript
import { Account } from "@formance/formance-sdk/sdk/models/payments";

let value: Account = {
  accountName: "<value>",
  connectorID: "<id>",
  createdAt: new Date("2026-01-17T18:22:24.834Z"),
  defaultAsset: "<value>",
  defaultCurrency: "<value>",
  id: "<id>",
  metadata: {},
  raw: {},
  reference: "<value>",
  type: "INTERNAL",
};
```
