## Example Usage

```typescript
import { BulkElement } from "@formance/formance-sdk/sdk/models/ledgerV3";

let value: BulkElement = {
  action: "ADD_METADATA",

    targetId: "<id>",
    targetType: "ACCOUNT",
  },
};
```
